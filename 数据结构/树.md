// =============================================================================
// 1. 双亲表示
// 结点放在连续的数组里。parent 存放双亲的下标，根的 parent 为 -1。
// 由下标找双亲是 O(1)；找一个结点的全部孩子要扫描整个数组。
// 目前类实现中只有第一讲的内容
// =============================================================================

struct ParentNode {
    char data;
    int parent;
};

class ParentTree {
public:
    vector<ParentNode> nodes;

    // 按下标取双亲。根没有双亲，得到的值是 -1。
    int parentOf(int i) const {
        return nodes[i].parent;  // 填空1：返回下标 i 的双亲下标。根没有双亲时，这个值是 -1
    }

    // 下标 i 的孩子，就是 parent 域等于 i 的那些结点。
    // 从头扫描到尾，得到的顺序与它们在数组中的先后一致。
    vector<int> children(int i) const {
        vector<int> result;
        for (int j = 0; j < static_cast<int>(nodes.size()); ++j) {
            if (node[j].parent == i) result.push_back(j);  // 填空2：parent 域等于 i 的结点，就是下标 i 的孩子
        }
        return result;
    }

    // 深度等于从该结点沿 parent 走到根所经过的边数。根的深度是 0。
    int depth(int i) const {
        int d = 0;
        while (nodes[i].parent != -1) {  // 填空3：当前结点还不是根，就继续沿 parent 向上
            i = nodes[i].parent;         // 填空4：改到双亲结点。下面的 ++d 把走过的边数加上 1
            ++d;
        }
        return d;
    }
};

// 按先根次序装入示例树。数组下标：A=0，B=1，C=2，D=3，E=4，F=5，G=6。
ParentTree demoParentTree() {
    ParentTree t;
    t.nodes = {
        {'A', -1}, {'B', 0}, {'C', 0}, {'D', 0},
        {'E', 1},  {'F', 1}, {'G', 3},  // 填空5：G 的双亲是 D。按上面的先根次序，D 的下标是 3
    };
    return t;
}

// =============================================================================
// 2. 孩子兄弟表示
// 每个结点两个指针：
//   firstChild   指向最左孩子（长子），没有孩子时为空
//   nextSibling  指向紧邻的右兄弟，没有右兄弟时为空
// 访问一个结点的全部孩子：先沿 firstChild 找到第一个，再沿 nextSibling 走到链尾。
// 结点里不再放 parent。找双亲用上面的 ParentTree。
// 第 2 讲将复用这里的类实现，两个指针对应成二叉树的左孩子、右孩子，类和结点都不再更换。
// =============================================================================

struct CsNode {
    char data;
    CsNode* firstChild;
    CsNode* nextSibling;
    explicit CsNode(char d) : data(d), firstChild(nullptr), nextSibling(nullptr) {}
};

class Tree {
public:
    Tree() : root_(nullptr) {} //构造函数 root_是类的私有成员，表示根节点指针
    ~Tree() { destroy(root_); } //析构函数 释放内存

    // 结点归这棵树释放。这里禁止拷贝，避免两个对象释放同一批结点。
    // 复制整棵树是第 3 次课的内容。
    Tree(const Tree&) = delete; //通过显式删除函数delete禁止拷贝
    Tree& operator=(const Tree&) = delete;

    bool empty() const { return root_ == nullptr; }
    CsNode* root() const { return root_; }

    // 空树时建立根。已有根时不再建立，返回空指针。
    CsNode* createRoot(char value);

    // 孩子个数，也就是该结点的度。沿兄弟链现数，不另存一个度数。
    int degree(CsNode* p) const;

    // 把新结点接到 parent 的孩子链末尾，使它成为最右的孩子。
    CsNode* insertChild(CsNode* parent, char value);

    // 删除 parent 的第 index 个孩子（从 0 计）以及该孩子的全部子孙。
    bool removeChild(CsNode* parent, int index);

    void printChildren(CsNode* parent) const;

    // 第 2 次课在这个类里继续添加成员，结点仍是 CsNode，根仍是 root_：

    static void destroy(CsNode* p);
};

    if (root_ != nullptr) return nullptr;
    root_ = new CsNode(value);
    return root_;
}

    if (p == nullptr) return 0;
    for (CsNode* c 
    = p->firstChild; c != nullptr; c = c->nextSibling) ++n;

}

    if (p == nullptr) return;

    delete p;
