```c++

#include <iostream>
using namespace std;

const int MaxTerms = 100;

// 1. 前置声明
template <typename Type>
class SparseMatrix;

// 2. 定义三元组
template <typename Type>
class Trituple {
    friend class SparseMatrix<Type>;

private:
    int row;
    int col;
    Type data;
};

// 3. 定义稀疏矩阵类
template <typename Type>
class SparseMatrix {
public:
    SparseMatrix(int maxRow, int maxCol); //声明构造函数
    bool AddTerm(int row, int col, Type value); //声明一个添加非零元素的成员函数，成功返回 `true`，失败返回 `false`。
    SparseMatrix Mat_Transpose() const; //声明普通转置函数，返回一个新的稀疏矩阵对象，不修改当前矩阵的普通成员状态。
    void Print() const;

private:
    int Rows;
    int Cols;
    int NonZero_Terms;
    Trituple<Type> SMArray[MaxTerms];
};

// 4. 实现构造函数
template <typename Type>
SparseMatrix<Type>::SparseMatrix(int maxRow, int maxCol) {
    Rows = maxRow;
    Cols = maxCol;
    NonZero_Terms = 0;
}

// 5. 实现函数：添加一个非零元素
template <typename Type>
bool SparseMatrix<Type>::AddTerm(int row, int col, Type value) {
    if (row < 0 || row >= Rows ||
        col < 0 || col >= Cols ||
        NonZero_Terms >= MaxTerms) {
        return false;
    }

    SMArray[NonZero_Terms].row = row;
    SMArray[NonZero_Terms].col = col;
    SMArray[NonZero_Terms].data = value;

    NonZero_Terms++;
    return true;
}

// 6. 普通转置算法（学校PPT的核心）
template <typename Type> //这个函数在类外实现，而且属于类模板，因此必须在这个定义前面声明类型模板参数。
SparseMatrix<Type> SparseMatrix<Type>::Mat_Transpose() const {
    SparseMatrix<Type> b(this->Cols, this->Rows);

    int CurrentPosition = 0;
    b.NonZero_Terms = this->NonZero_Terms;

    for (int k = 0; k < Cols; k++) {
        for (int i = 0; i < NonZero_Terms; i++) {
            if (SMArray[i].col == k) {
                b.SMArray[CurrentPosition].row = k;
                b.SMArray[CurrentPosition].col = this->SMArray[i].row;
                b.SMArray[CurrentPosition].data = this->SMArray[i].data;

                CurrentPosition++;
            }
        }
    }

    return b;
}

// 7. 输出三元组
template <typename Type>
void SparseMatrix<Type>::Print() const {
    cout << "Rows=" << Rows
         << ", Cols=" << Cols
         << ", NonZero=" << NonZero_Terms << endl;

    for (int i = 0; i < NonZero_Terms; i++) {
        cout << "("
             << SMArray[i].row << ","
             << SMArray[i].col << ","
             << SMArray[i].data
             << ")" << endl;
    }
}

全局函数的调用方法： Add(10, 20);
成员函数的调用方法：A.Mat_Transpose();
```

```c++
快速转置算法：
目标：避免反复遍历原数组，只遍历一次，同时保证新三元组行优先储存
方法：利用数组的随机访问，已知转置后每个元素在数组中存放的位置直接访问
rowSize = [1,  0, 2, 1];存放转置后每行有多少个非零元素
rowStart= [0, 1, 1, 3];存放每一行从什么下标开始

快速转置利用辅助数组 rowSize 和 rowStart 确定各行的存储区域，并利用原三元组表按行号优先排列的性质，保证转置后同一行内的元素按列号升序排列。
```

```c++

#include <iostream>
using namespace std;

const int MaxTerms = 100;

// 提前声明稀疏矩阵类模板
template <class Type>
class SparseMatrix;

// 定义三元组类模板
template <class Type>
class Trituple {
    friend class SparseMatrix<Type>;

private:
    int row, col;
    Type data;
};

// 定义稀疏矩阵类模板
template <class Type>
class SparseMatrix {
public:
    SparseMatrix(int MaxRow, int MaxCol);
    bool AddTerm(int row, int col, Type value);
    SparseMatrix Mat_Fast_Transpose() const;
    void Print() const;

private:
    int Rows;
    int Cols;
    int NonZero_Terms;
    Trituple<Type> SMArray[MaxTerms];
};

// 构造函数
template <class Type>
SparseMatrix<Type>::SparseMatrix(int MaxRow, int MaxCol) {
    Rows = MaxRow;
    Cols = MaxCol;
    NonZero_Terms = 0;
}

// 添加非零元素
template <class Type>
bool SparseMatrix<Type>::AddTerm(int row, int col, Type value) {
    if (row < 0 || row >= Rows ||
        col < 0 || col >= Cols ||
        NonZero_Terms >= MaxTerms) {
        return false;
    }

    // 保证三元组按行号、列号升序添加
    if (NonZero_Terms > 0) {
        int oldRow = SMArray[NonZero_Terms - 1].row;
        int oldCol = SMArray[NonZero_Terms - 1].col;

        if (row < oldRow ||
            (row == oldRow && col <= oldCol)) {
            return false;
        }
    }

    SMArray[NonZero_Terms].row = row;
    SMArray[NonZero_Terms].col = col;
    SMArray[NonZero_Terms].data = value;

    NonZero_Terms++;
    return true;
}

// 快速转置（学校PPT核心算法）
template <class Type>
SparseMatrix<Type> SparseMatrix<Type>::Mat_Fast_Transpose() const {
    SparseMatrix<Type> b(Cols, Rows);

    int i, j;//i用于遍历数组，j用于保存新数组中的目标下标
    int* x = new int[Cols];
    int* y = new int[Cols];

    b.NonZero_Terms = NonZero_Terms;

    if (NonZero_Terms > 0) {
        for (i = 0; i < Cols; i++) {
            x[i] = 0;
        }

        for (i = 0; i < NonZero_Terms; i++) { //初始化x数组，方便后面用作计数器
            x[SMArray[i].col]++; //用当前三元组的列号作为 x 数组的下标。
        }

        y[0] = 0;

        for (i = 1; i < Cols; i++) {
            y[i] = y[i - 1] + x[i - 1];
        } 
		//正式转置
        for (i = 0; i < NonZero_Terms; i++) {
            j = y[SMArray[i].col]; //查出这个三元组转置后应该写入的位置，把位置保存到 j

            b.SMArray[j].row = SMArray[i].col;
            b.SMArray[j].col = SMArray[i].row;
            b.SMArray[j].data = SMArray[i].data;

            y[SMArray[i].col]++;
        }
    }

    delete[] x;
    delete[] y;

    return b;
}

// 输出三元组
template <class Type>
void SparseMatrix<Type>::Print() const {
    cout << "Rows=" << Rows
         << ", Cols=" << Cols
         << ", NonZero_Terms=" << NonZero_Terms << endl;

    for (int i = 0; i < NonZero_Terms; i++) {
        cout << "("
             << SMArray[i].row << ", "
             << SMArray[i].col << ", "
             << SMArray[i].data << ")" << endl;
    }
}

int main() {
    // 创建3行4列的矩阵
    SparseMatrix<int> A(3, 4);

    // 添加非零元素
    A.AddTerm(0, 2, 5);
    A.AddTerm(1, 0, 7);
    A.AddTerm(1, 2, 8);
    A.AddTerm(2, 3, 9);

    cout << "A:" << endl;
    A.Print();

    // 执行快速转置
    SparseMatrix<int> B = A.Mat_Fast_Transpose();

    cout << "A transpose:" << endl;
    B.Print();

    return 0;
}
```
