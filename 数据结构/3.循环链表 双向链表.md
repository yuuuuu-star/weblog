1.顺序表版本
```c++
#include <iostream>
using namespace std;

const int MAXSIZE = 100;

// 定义顺序表
struct SqList {
    int data[MAXSIZE];
    int length;
};

// 初始化顺序表
void InitList(SqList &L) {
    L.length = 0;
}

// 按值查找，找不到返回0
int LocateElem(const SqList &L, int e) {
    for (int i = 0; i < L.length; i++) {
        if (L.data[i] == e)
            return i + 1;
    }
    return 0;
}

// 在第i个位置插入元素e，位置从1开始
bool ListInsert(SqList &L, int i, int e) {
    if (L.length >= MAXSIZE || i < 1 || i > L.length + 1) {
        return false;
    }

    // 从后向前移动元素
    for (int j = L.length - 1; j >= i - 1; j--) {
        L.data[j + 1] = L.data[j];
    }

    L.data[i - 1] = e;
    L.length++;
    return true;
}

// 集合式合并：把Lb中La没有的元素追加到La
bool UnionSet(SqList &La, const SqList &Lb) {
    for (int i = 0; i < Lb.length; i++) {
        int e = Lb.data[i];

        if (LocateElem(La, e) == 0) {
            if (!ListInsert(La, La.length + 1, e))
                return false;
        }
    }
    return true;
}

// 输出顺序表
void PrintList(const SqList &L) {
    for (int i = 0; i < L.length; i++) {
        cout << L.data[i] << " ";
    }
    cout << endl;
}

int main() {
    SqList La, Lb;
    InitList(La);
    InitList(Lb);

    int m, n;

    cout << "请输入La的长度：";
    if (!(cin >> m) || m < 0 || m > MAXSIZE)
        return 1;

    cout << "请输入La的元素：";
    for (int i = 0; i < m; i++) {
        int x;
        if (!(cin >> x) || !ListInsert(La, La.length + 1, x))
            return 1;
    }

    cout << "请输入Lb的长度：";
    if (!(cin >> n) || n < 0 || n > MAXSIZE)
        return 1;

    cout << "请输入Lb的元素：";
    for (int i = 0; i < n; i++) {
        int x;
        if (!(cin >> x) || !ListInsert(Lb, Lb.length + 1, x))
            return 1;
    }

    if (UnionSet(La, Lb)) {
        cout << "集合式合并结果：";
        PrintList(La);
    } else {
        cout << "合并失败：顺序表容量不足" << endl;
    }

    return 0;
}
```

```c++
//先遍历Lb，依次取出其中的每个元素，操作如下

#include <iostream>
using namespace std;

// 定义单链表
struct LNode {
    int data;
    LNode *next;
};

typedef LNode* LinkList;

// 尾插法创建单链表
void CreateList(LinkList &L, int n) {
    L = new LNode;
    L->next = nullptr;

    LNode *r = L;

    for (int i = 0; i < n; i++) {
        LNode *p = new LNode;
        cin >> p->data;
        p->next = nullptr;

        r->next = p;
        r = p;
    }
}

// 查找链表中是否存在元素e
bool Contains(LinkList L, int e) {
    LNode *p = L->next;

    while (p != nullptr) {
        if (p->data == e)
            return true;

        p = p->next;
    }
    return false;
}

// 集合式合并
void UnionSet(LinkList La, LinkList Lb) {
    LNode *r = La;

    // 让r指向La的尾结点
    while (r->next != nullptr) {
        r = r->next;
    }

    // 遍历Lb
    LNode *p = Lb->next;

    while (p != nullptr) {
        // 如果La中不存在当前元素
        if (!Contains(La, p->data)) {
            LNode *q = new LNode;
            q->data = p->data;
            q->next = nullptr;

            // 尾插到La中
            r->next = q;
            r = q;
        }

        p = p->next;
    }
}

// 输出链表
void PrintList(LinkList L) {
    LNode *p = L->next;

    while (p != nullptr) {
        cout << p->data << " ";
        p = p->next;
    }
    cout << endl;
}

// 释放链表
void DestroyList(LinkList &L) {
    while (L != nullptr) {
        LNode *p = L;
        L = L->next;
        delete p;
    }
}

int main() {
    LinkList La, Lb;
    int m, n;

    cout << "请输入La的长度：";
    if (!(cin >> m) || m < 0)
        return 1;

    cout << "请输入La的元素：";
    CreateList(La, m);

    cout << "请输入Lb的长度：";
    if (!(cin >> n) || n < 0) {
        DestroyList(La);
        return 1;
    }

    cout << "请输入Lb的元素：";
    CreateList(Lb, n);

    UnionSet(La, Lb);

    cout << "集合式合并结果：";
    PrintList(La);

    DestroyList(La);
    DestroyList(Lb);

    return 0;
}
```

```c++
#include <iostream>
using namespace std;

const int MAXSIZE = 100;

// 定义顺序表
struct SqList {
    int data[MAXSIZE];
    int length;
};

// 合并两个升序顺序表
bool MergeList(SqList La, SqList Lb, SqList &Lc) {
    if (La.length + Lb.length > MAXSIZE) {
        return false;
    }

    int i = 0, j = 0, k = 0;

    // 两个表都有未处理元素时，比较大小
    while (i < La.length && j < Lb.length) {
        if (La.data[i] <= Lb.data[j]) {
            Lc.data[k] = La.data[i];
            i++;
        } else {
            Lc.data[k] = Lb.data[j];
            j++;
        }
        k++;
    }

    // La 还有剩余元素
    while (i < La.length) {
        Lc.data[k] = La.data[i];
        i++;
        k++;
    }

    // Lb 还有剩余元素
    while (j < Lb.length) {
        Lc.data[k] = Lb.data[j];
        j++;
        k++;
    }

    Lc.length = k;
    return true;
}

int main() {
    SqList La, Lb, Lc;

    // 输入第一个顺序表
    cout << "请输入La的长度：";
    cin >> La.length;

    if (La.length < 0 || La.length > MAXSIZE)
        return 1;

    cout << "请输入La的元素（升序）：";
    for (int i = 0; i < La.length; i++) {
        cin >> La.data[i];
    }

    // 输入第二个顺序表
    cout << "请输入Lb的长度：";
    cin >> Lb.length;

    if (Lb.length < 0 || Lb.length > MAXSIZE)
        return 1;

    cout << "请输入Lb的元素（升序）：";
    for (int i = 0; i < Lb.length; i++) {
        cin >> Lb.data[i];
    }

    if (!MergeList(La, Lb, Lc)) {
        cout << "合并失败：空间不足" << endl;
        return 1;
    }

    cout << "合并结果：";
    for (int i = 0; i < Lc.length; i++) {
        cout << Lc.data[i] << " ";
    }
    cout << endl;

    return 0;
}
```

```c++

#include <iostream>
using namespace std;

// 定义单链表结点
struct LNode {
    int data;
    LNode *next;
};

typedef LNode* LinkList;

// 尾插法创建带头结点的单链表
void CreateList(LinkList &L, int n) {
    L = new LNode;
    L->next = nullptr;

    LNode *r = L;

    for (int i = 0; i < n; i++) {
        LNode *p = new LNode;
        cin >> p->data;
        p->next = nullptr;

        r->next = p;
        r = p;
    }
}

// 合并两个升序单链表，结果保存在La中
void MergeList(LinkList &La, LinkList &Lb) {
    LNode *pa = La->next;
    LNode *pb = Lb->next;
    LNode *r = La;

    // 两个链表均有剩余结点
    while (pa != nullptr && pb != nullptr) {
        if (pa->data <= pb->data) {
            r->next = pa;
            r = pa;
            pa = pa->next;
        } else {
            r->next = pb;
            r = pb;
            pb = pb->next;
        }
    }

    // 把尚未处理的剩余结点接到尾部
    if (pa != nullptr) {
        r->next = pa;
    } else {
        r->next = pb;
    }

    // Lb的头结点已经不再需要
    delete Lb;
    Lb = nullptr;
}

// 输出链表
void PrintList(LinkList L) {
    LNode *p = L->next;

    while (p != nullptr) {
        cout << p->data << " ";
        p = p->next;
    }
    cout << endl;
}

// 释放链表占用的内存
void DestroyList(LinkList &L) {
    while (L != nullptr) {
        LNode *p = L;
        L = L->next;
        delete p;
    }
}

int main() {
    LinkList La, Lb;
    int m, n;

    cout << "请输入La的长度：";
    cin >> m;
    if (m < 0) return 1;

    cout << "请输入La的元素（升序）：";
    CreateList(La, m);

    cout << "请输入Lb的长度：";
    cin >> n;
    if (n < 0) {
        DestroyList(La);
        return 1;
    }

    cout << "请输入Lb的元素（升序）：";
    CreateList(Lb, n);

    MergeList(La, Lb);

    cout << "合并结果：";
    PrintList(La);

    DestroyList(La);

    return 0;
}
```
